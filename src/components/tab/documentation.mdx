props:
  - name: badgeLabel
    type: string
    required: false
    description: Shows a Badge on this tab. Pass a count (a number over 999 shows as `999+`, and `"0"` hides the badge) or `""` for a dot. On a primary tab with an `icon`, the badge overlaps the icon; otherwise it sits after the label. A count is added to the tab's accessible name (e.g. "Inbox, 3"), since the badge itself is hidden from screen readers. A dot adds nothing to the name, so say what it means in the tab's content or label.
  - name: icon
    type: SvgIconNameTypes
    required: false
    description: An icon shown with this tab's label - above it on primary tabs, before it on secondary tabs.
  - name: label
    type: string
    required: true
    description: The text shown on this tab's button in the tab list.
  - name: transition
    type: '"fade" | "fade-up" | "slide-left"'
    required: false
    default: '"fade"'
    description: Which transition to play when this tab's panel swaps into view. The `fade` transition is a simple fade-in/fade-out, while the `fade-up` animates the panel fading in from below, and the `slide-left` animates the panel fading in and sliding from the right.
slots:
  - name: default
    description: The content to display when this tab is selected.
    kind: any
