category: Text content
role: pattern
meta:
  title: Tab | astro-components HenriFournier.dev
  description: A single tab panel within a Tabs component, holding one tab's label and content.
extends:
  kind: element
  tag: [div]
parents: [Tabs]
props:
  - name: badgeDescription
    type: string
    required: false
    default: '"New notification"'
    description: What a dot badge (`badgeLabel=""`) means, added after the label in the tab's accessible name, as M3's "New notification" is. Set it to translate the phrase or to name another status, e.g. "New draft". It has no effect on a count.
  - name: badgeLabel
    type: string
    required: false
    description: Shows a Badge on this tab. Pass a count (a number over 999 shows as `999+`, and `"0"` hides the badge) or `""` for a dot. On a primary tab with an `icon`, the badge overlaps the icon; otherwise it sits after the label. The badge itself is hidden from screen readers, so, as in M3, a count is added to the tab's accessible name (e.g. "Inbox, 3"), and a dot adds `badgeDescription` ("Updates, New notification").
  - name: icon
    type: SvgIconNameTypes
    required: false
    description: An icon shown with this tab's label - above it on primary tabs, before it on secondary tabs.
  - name: label
    type: string
    required: true
    description: The text shown on this tab's button in the tab list.
  - name: transition
    type: '"fade" | "fade-up" | "slide-left"'
    required: false
    default: '"fade"'
    description: Which transition to play when this tab's panel swaps into view. The `fade` transition is a simple fade-in/fade-out, while the `fade-up` animates the panel fading in from below, and the `slide-left` animates the panel fading in and sliding from the right. Any other value fails the build.
slots:
  - name: default
    description: The content to display when this tab is selected.
    kind: any
---
